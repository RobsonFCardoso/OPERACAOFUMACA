<script>
  const SCRIPT_URL = "https://script.google.com/macros/s/AKfycbyiVUz1UVYlqk2uH93-5BrUJXdA3oIPfqYV23Y0lA6xjLcDIa8WY_zOsziveQxaP8fmDQ/exec";
  let textoMensagemGlobal = "";

  function fileToBase64(file) {
    return new Promise((resolve, reject) => {
      const reader = new FileReader();
      reader.onload = () => resolve(reader.result);
      reader.onerror = error => reject(error);
      reader.readAsDataURL(file);
    });
  }

  const btnSubmit = document.getElementById('btnSubmit');

  btnSubmit.addEventListener('click', async function(e) {
    e.preventDefault();

    const idOperacao = document.getElementById('idOperacao').value.trim();
    const blitz = document.getElementById('blitz').value.trim();
    const veiculo = document.getElementById('veiculo').value.trim();
    const placa = document.getElementById('placa').value.trim();
    const status = document.getElementById('status').value;

    if (!idOperacao || !blitz || !veiculo || !placa || !status) {
      alert("Por favor, preencha todos os campos obrigatórios (*).");
      return;
    }

    btnSubmit.disabled = true;
    btnSubmit.textContent = "Salvando fotos e dados...";

    async function processFile(inputId) {
      const input = document.getElementById(inputId);
      if (input.files.length > 0) {
        const file = input.files[0];
        return {
          name: file.name,
          mimeType: file.type,
          data: await fileToBase64(file)
        };
      }
      return null;
    }

    const payload = {
      idOperacao: idOperacao,
      blitz: blitz,
      veiculo: veiculo,
      placa: placa,
      status: status,
      fotoFrente: await processFile('fotoFrente'),
      fotoTraz: await processFile('fotoTraz'),
      fotoCnh: await processFile('fotoCnh'),
      fotoCrlvF: await processFile('fotoCrlvF'),
      fotoCrlvV: await processFile('fotoCrlvV')
    };

    try {
      await fetch(SCRIPT_URL, {
        method: 'POST',
        mode: 'no-cors',
        headers: {
          'Content-Type': 'text/plain'
        },
        body: JSON.stringify(payload)
      });

      var dataHoje = new Date();
      var dataStr = dataHoje.toISOString().split('T')[0];
      var horaStr = dataHoje.toTimeString().substring(0, 5);

      var emojiStatus = "⏳";
      if (status === "APROVADO") emojiStatus = "✅";
      else if (status === "REPROVADO") emojiStatus = "❌";
      else if (status === "LIBERADO") emojiStatus = "🟢";

      textoMensagemGlobal = "🚗 *RELATÓRIO DE VISTORIA VEICULAR*\n" +
                            "_____________\n\n" +
                            "📋 *ID do Registro:* " + idOperacao + "\n" +
                            "🛡️ *Nome Blitz:* " + blitz + "\n" +
                            "🛞 *Placa:* " + placa.toUpperCase() + "\n" +
                            "📅 *Data:* " + dataStr + "\n" +
                            "⏰ *Hora:* " + horaStr + "\n" +
                            emojiStatus + " *Status:* " + status + "\n" +
                            "_____________\n\n" +
                            "_Emitido via Sistema de Vistorias e Registros._";

      var linkWsp = "https://api.whatsapp.com/send?phone=5511974987081&text=" + encodeURIComponent(textoMensagemGlobal);

      document.getElementById('formCard').classList.add('hidden');
      document.getElementById('whatsappLink').href = linkWsp; 
      document.getElementById('successCard').classList.remove('hidden');

    } catch (err) {
      alert("Erro na comunicação: " + err);
      btnSubmit.disabled = false;
      btnSubmit.textContent = "Salvar Vistoria e Gerar WhatsApp";
    }
  });

  document.getElementById('whatsappLink').addEventListener('click', function() {
    navigator.clipboard.writeText(textoMensagemGlobal).catch(err => {
      console.log('Erro ao copiar texto: ', err);
    });
  });

  document.getElementById('btnVoltar').addEventListener('click', function() {
    document.getElementById('vistoriaForm').reset();
    btnSubmit.disabled = false;
    btnSubmit.textContent = "Salvar Vistoria e Gerar WhatsApp";

    document.getElementById('successCard').classList.add('hidden');
    document.getElementById('formCard').classList.remove('hidden');
  });
</script>
